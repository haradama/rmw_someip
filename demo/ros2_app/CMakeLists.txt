cmake_minimum_required(VERSION 3.16)
project(rmw_someip_demo_ros2_app CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rmw_someip_demo_msgs REQUIRED)
add_executable(vehicle_node src/main.cpp)
target_link_libraries(vehicle_node rclcpp::rclcpp ${rmw_someip_demo_msgs_TARGETS})
install(TARGETS vehicle_node DESTINATION lib/${PROJECT_NAME})
ament_package()
