cmake_minimum_required(VERSION 3.16)
project(pubsub_smoke CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(pubsub_smoke main.cpp)
target_link_libraries(pubsub_smoke rclcpp::rclcpp ${std_msgs_TARGETS})
install(TARGETS pubsub_smoke DESTINATION lib/${PROJECT_NAME})
ament_package()
